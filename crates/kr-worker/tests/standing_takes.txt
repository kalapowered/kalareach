# The takes. Each is a lock of one domain, or the session's guard.
# The time contract's state lock: the proven reading's mark is read under it, and nothing else.
site crates/kr-worker/src/action/time.rs pure_proven_ms lock 1
# The gate's own lock: the bound is copied out and the lock is let go before any clock is read.
site crates/kr-worker/src/local_gate.rs check_bound lock 1
# The authority state.
site crates/kr-worker/src/service.rs check_authority lock 1
# The input domain's lock, for the revision the request was validated under.
site crates/kr-worker/src/service.rs check_validated_revision lock 1
# The authority state and the input domain's lock, each taken for one read.
site crates/kr-worker/src/service.rs check_dispatch_lease lock 2
# The session, for the id and the epoch the target is compared with, let go before the next check.
site crates/kr-worker/src/service.rs check_envelope session 1
# The session, for the journal's two reads (taken here only when no pass holds it), and the lock of
# the record of what a person was shown, which is read once and let go.
site crates/kr-worker/src/service.rs check_admission_limits session 1
site crates/kr-worker/src/service.rs check_admission_limits lock 1
# `Session::state()` is a read of the session the caller holds, not a take.
site crates/kr-worker/src/service.rs check_preconditions state 2
